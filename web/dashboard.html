Quick setup

1. Install
npm install sphegatti
2. Create ota.config.js (or run npx ota init)
module.exports = {
  appId: 'your-app',
  runtimeVersion: '1',
  publish: {
    serverUrl: 'https://sphegatti.onrender.com',
    apiKey: 'YOUR_API_KEY',
  },
};
3. Publish an update
npx ota publish --platform android --channel production
4. Check for updates in your app (React Native)
import { NativeModules } from 'react-native';
const { OtaModule } = NativeModules;

const res = await fetch(
  'https://sphegatti.onrender.com/updates/check?' +
  new URLSearchParams({ appId, platform, runtimeVersion, channel, currentUpdateId })
);
const { updateAvailable, bundleUrl } = await res.json();
if (updateAvailable) await OtaModule.downloadAndActivate(bundleUrl);
